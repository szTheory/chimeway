```elixir
defmodule Chimeway.Test.DispatchHelpers do
  @moduledoc "Factory helpers for Phase 3 async dispatch tests."

  alias Chimeway.{Deliveries, Repo}

  @doc "Creates event + notification + delivery in a ready-to-dispatch state."
  def create_pending_delivery(opts \\ []) do
    notification_key = Keyword.get(opts, :notification_key, "test_notifier")
    recipient_identity = Keyword.get(opts, :recipient_identity, "user:#{System.unique_integer()}")
    channel = Keyword.get(opts, :channel, :in_app)
    delay_fallback = Keyword.get(opts, :delay_fallback, false)

    {:ok, event} =
      Repo.insert(%Chimeway.Events.Event{
        notification_key: notification_key,
        notification_version: 1,
        idempotency_key: "test-#{System.unique_integer()}",
        payload: %{}
      })

    {:ok, notification} =
      Repo.insert(%Chimeway.Notifications.Notification{
        event_id: event.id,
        recipient_identity: recipient_identity,
        recipient_type: "user",
        metadata: %{}
      })

    {:ok, delivery} =
      %Chimeway.Delivery{}
      |> Chimeway.Delivery.changeset(%{
        notification_id: notification.id,
        channel: to_string(channel),
        status: :pending,
        delay_fallback: delay_fallback
      })
      |> Repo.insert()

    %{event: event, notification: notification, delivery: delivery}
  end

  @doc "Sets read_at on a notification to simulate in-app read."
  def mark_notification_read(%{notification: notification}) do
    {:ok, updated} =
      notification
      |> Ecto.Changeset.change(read_at: DateTime.utc_now() |> DateTime.truncate(:microsecond))
      |> Repo.update()
    updated
  end
end
```

```elixir
  alias Chimeway.DeliveryAttempt

    :ok
  end

    test "records exactly one attempt row and transitions delivery to :succeeded" do
      %{delivery: delivery} = create_pending_delivery()

      assert :ok = perform_job(ObanWorker, %{delivery_id: delivery.id})

      updated = Deliveries.get_delivery!(delivery.id)
      assert updated.status == :succeeded

      attempts = Chimeway.Repo.all(
        Ecto.Query.from(a in DeliveryAttempt, where: a.delivery_id == ^delivery.id)
      )
      assert length(attempts) == 1
      assert hd(attempts).outcome == :succeeded
    end
  end

  describe "perform/1 — idempotency under retry" do
    test "running perform twice creates exactly one attempt row" do
      %{delivery: delivery} = create_pending_delivery()

      assert :ok = perform_job(ObanWorker, %{delivery_id: delivery.id})

      assert :ok = perform_job(ObanWorker, %{delivery_id: delivery.id})

      attempts = Chimeway.Repo.all(

  end
```

```elixir
defmodule Chimeway.Policy.DelayedFallbackTest do
  use Chimeway.DataCase, async: false
  use Oban.Testing, repo: Chimeway.Repo

  @moduletag :oban
  @moduletag :policy

  import Chimeway.Test.DispatchHelpers

  alias Chimeway.{Deliveries, Dispatch.ObanWorker, Preferences}

  setup do
    Application.put_env(:chimeway, :adapter, Chimeway.Adapters.Test)
    Chimeway.Adapters.Test.clear()
    on_exit(fn -> Application.put_env(:chimeway, :adapter, Chimeway.Adapters.Logger) end)
    :ok
  end

  describe "delay_fallback: true — notification NOT read" do
    test "adapter is called and delivery succeeds" do
      %{delivery: delivery} = create_pending_delivery(delay_fallback: true)

      assert :ok = perform_job(ObanWorker, %{delivery_id: delivery.id})

      updated = Deliveries.get_delivery!(delivery.id)
      assert updated.status == :succeeded
      assert updated.suppression_reason == nil
      refute Chimeway.Adapters.Test.delivered_messages() == []
    end
  end

  describe "delay_fallback: true — notification IS read" do
    test "delivery is suppressed with :already_read, adapter is NOT called" do
      ctx = create_pending_delivery(delay_fallback: true)
      _read_notification = mark_notification_read(ctx)
      Chimeway.Adapters.Test.clear()

      assert :ok = perform_job(ObanWorker, %{delivery_id: ctx.delivery.id})

      updated = Deliveries.get_delivery!(ctx.delivery.id)
      assert updated.status == :suppressed
      assert updated.suppression_reason == "already_read"
      assert Chimeway.Adapters.Test.delivered_messages() == []
    end

    test "suppressed delivery has zero attempt rows" do
      ctx = create_pending_delivery(delay_fallback: true)
      mark_notification_read(ctx)

      perform_job(ObanWorker, %{delivery_id: ctx.delivery.id})

      attempt_count = Chimeway.Repo.aggregate(
        Ecto.Query.from(a in Chimeway.DeliveryAttempt, where: a.delivery_id == ^ctx.delivery.id),
        :count
      )
      assert attempt_count == 0
    end
  end

  describe "delay_fallback: true — preference disabled after enqueue" do
    test "delivery is suppressed at perform time with channel_disabled reason" do
      ctx = create_pending_delivery(delay_fallback: true)

      # Disable preference after delivery is planned (simulates post-enqueue change)
      Preferences.upsert_preference(%{
        recipient_id: ctx.notification.recipient_identity,
        notification_key: ctx.event.notification_key,
        channel: ctx.delivery.channel,
        enabled: false
      })
      Chimeway.Adapters.Test.clear()

      assert :ok = perform_job(ObanWorker, %{delivery_id: ctx.delivery.id})

      updated = Deliveries.get_delivery!(ctx.delivery.id)
      assert updated.status == :suppressed
      assert updated.suppression_reason == "channel_disabled"
      assert Chimeway.Adapters.Test.delivered_messages() == []
    end
  end

  describe "sync dispatcher — delay_fallback: true with read state" do
    test "sync path also suppresses when notification is read (POLC-02)" do
      ctx = create_pending_delivery(delay_fallback: true)
      # Reload delivery with delay_fallback set
      delivery = Deliveries.get_delivery!(ctx.delivery.id)
      mark_notification_read(ctx)
      Chimeway.Adapters.Test.clear()

      # Simulate what Dispatch.Sync does: call dispatch_delivery via the policy check
      # Since dispatch_delivery is private, test via the full Dispatch.Sync.dispatch/2
      # with a notification that has the read delivery
      {:ok, _deliveries} = Chimeway.Dispatch.Sync.dispatch([ctx.notification], [])

      # The delivery was already pending — Sync will evaluate policy and suppress it
      updated = Deliveries.get_delivery!(delivery.id)
      assert updated.status == :suppressed
      assert updated.suppression_reason == "already_read"
      assert Chimeway.Adapters.Test.delivered_messages() == []
    end
  end
end
```

```elixir
defmodule Chimeway.Dispatch.ObanTransactionalTest do
  use Chimeway.DataCase, async: false
  use Oban.Testing, repo: Chimeway.Repo

  @moduletag :oban
  @moduletag :integration

  import Chimeway.Test.DispatchHelpers

  alias Chimeway.{Deliveries, Dispatch.Oban, Dispatch.ObanWorker, Repo}

  setup do
    Application.put_env(:chimeway, :adapter, Chimeway.Adapters.Test)
    Application.put_env(:chimeway, :dispatcher, Chimeway.Dispatch.Oban)
    on_exit(fn ->
      Application.put_env(:chimeway, :adapter, Chimeway.Adapters.Logger)
      Application.put_env(:chimeway, :dispatcher, Chimeway.Dispatch.Sync)
    end)
    :ok
  end

  describe "transactional enqueue — commit" do
    test "committed multi makes both delivery row and Oban job visible" do
      %{notification: notification} = create_pending_delivery()
      {:ok, delivery} = Deliveries.plan_delivery(notification.id, :in_app)

      # Create a multi that succeeds, inserting the job transactionally
      multi =
        Ecto.Multi.new()
        |> Ecto.Multi.run(:marker, fn _repo, _changes -> {:ok, :success} end)

      Oban.dispatch([notification], multi: multi)

      assert_enqueued(worker: ObanWorker, args: %{delivery_id: delivery.id})
    end
  end

  describe "transactional enqueue — rollback" do
    test "rolled-back multi removes Oban job from queue" do
      %{notification: notification} = create_pending_delivery()
      {:ok, delivery} = Deliveries.plan_delivery(notification.id, :in_app)

      failing_multi =
        Ecto.Multi.new()
        |> Ecto.Multi.run(:fail, fn _repo, _changes -> {:error, :forced_failure} end)

      Oban.dispatch([notification], multi: failing_multi)

      refute_enqueued(worker: ObanWorker, args: %{delivery_id: delivery.id})
    end
  end

  describe "idempotency under duplicate dispatch" do
    test "dispatching the same notification twice enqueues one job (plan_delivery idempotency)" do
      %{notification: notification} = create_pending_delivery()

      Oban.dispatch([notification], [])
      Oban.dispatch([notification], [])

      # Only one delivery row exists for this notification+channel
      delivery_count = Repo.aggregate(
        Ecto.Query.from(d in Chimeway.Delivery, where: d.notification_id == ^notification.id),
        :count
      )
      assert delivery_count == 1
    end
  end

  describe "Chimeway.Dispatch.Sync — no Oban jobs" do
    test "sync dispatcher config produces zero Oban job inserts" do
      Application.put_env(:chimeway, :dispatcher, Chimeway.Dispatch.Sync)
      %{notification: notification} = create_pending_delivery()

      Chimeway.Dispatch.Sync.dispatch([notification], [])

      assert all_enqueued(worker: ObanWorker) == []
    after
      Application.put_env(:chimeway, :dispatcher, Chimeway.Dispatch.Oban)
    end
  end
end
```

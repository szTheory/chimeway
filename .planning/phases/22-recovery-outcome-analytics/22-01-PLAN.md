def list_due_deferred_deliveries(opts \\ []) when is_list(opts) do
  Repo.all(
    from(d in Delivery,
      where:
        d.status == :pending and d.orchestration_state == :deferred and
          not is_nil(d.next_eligible_at) and d.next_eligible_at <= ^now
    )
  )
end

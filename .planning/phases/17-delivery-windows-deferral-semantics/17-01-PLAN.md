```elixir
def upsert_settings(attrs) do
  %Setting{}
  |> Setting.changeset(attrs)
  |> Repo.insert(
    on_conflict:
      {:replace,
       [
         :quiet_hours_start_minute,
         :quiet_hours_end_minute,
         :delivery_cap_count,
         :delivery_cap_window_minutes,
         :updated_at
       ]},
    conflict_target: [:recipient_id]
  )
end
```

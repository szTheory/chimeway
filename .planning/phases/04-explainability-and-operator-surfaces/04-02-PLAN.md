```elixir
defmodule Chimeway.Telemetry do
  @moduledoc """
  Instrumentation facade for all Chimeway telemetry.

  Wraps `:telemetry.span/3` with the `[:chimeway | suffix]` prefix and enforces
  structural PII redaction via `safe_meta/1` before any metadata is emitted.

  ## Attaching the default Logger handler

  Add to your `Application.start/2`:

      def start(_type, _args) do
        Chimeway.Telemetry.attach_default_handlers()
        # ...
      end

  The default handler logs `:stop` events at `:debug` level and `:exception`
  events at `:warning` level. It is safe to call multiple times.

  ## Telemetry Event Catalog

  All events follow the `[:chimeway, domain, action, :start | :stop | :exception]`
  convention.

  | Event name | When emitted | Allowed metadata keys |
  |---|---|---|
  | `[:chimeway, :events, :create]` | Event row inserted in trigger pipeline | notification_key, event_id, correlation_id |
  | `[:chimeway, :deliveries, :plan]` | Delivery rows planned per recipient | notification_key, event_id |
  | `[:chimeway, :policy, :evaluate]` | Policy evaluated for a single delivery | delivery_id, channel, outcome, suppression_reason |
  | `[:chimeway, :dispatch, :sync]` | Sync adapter call per delivery | delivery_id, channel, outcome |
  | `[:chimeway, :dispatch, :enqueue]` | Oban job enqueued per delivery (Oban only) | delivery_id |
  | `[:chimeway, :dispatch, :perform]` | Oban worker performs a delivery (Oban only) | delivery_id, channel, outcome |
  | `[:chimeway, :attempts, :record]` | Attempt row recorded after adapter call | delivery_id, outcome |

  ## PII Redaction

  `safe_meta/1` is the enforcement boundary. It accepts any map and returns only
  the keys in the allowed set, silently dropping all others.

  Allowed keys: `[:notification_key, :event_id, :recipient_id, :channel, :delivery_id,
  :attempt_id, :outcome, :suppression_reason, :correlation_id]`

  ## Writing a custom handler

      :telemetry.attach_many(
        :my_app_chimeway,
        [[:chimeway, :events, :create, :stop], [:chimeway, :attempts, :record, :stop]],
        fn event, measurements, metadata, _config ->
          MyApp.Metrics.increment(event, metadata)
        end,
        nil
      )
  """

  require Logger

  @allowed_meta_keys ~w(
    notification_key
    event_id
    recipient_id
    channel
    delivery_id
    attempt_id
    outcome
    suppression_reason
    correlation_id
  )a

  @doc """
  Emits a :telemetry.span/3 event under the [:chimeway | event_suffix] prefix.

  func must return {result, extra_stop_meta}. Pass safe_meta/1 on the meta map
  and on extra stop metadata to enforce PII redaction.
  """
  @spec span([atom()], map(), (-> {term(), map()})) :: term()
  def span(event_suffix, meta, func) do
    :telemetry.span([:chimeway | event_suffix], meta, func)
  end

  @doc """
  Returns a new map containing only the allowed telemetry metadata keys.

  Handles both atom and string keys — normalises to atom before filtering.
  All other keys are silently dropped.

      iex> Chimeway.Telemetry.safe_meta(%{notification_key: "k", email: "u@example.com"})
      %{notification_key: "k"}
  """
  @spec safe_meta(map()) :: map()
  def safe_meta(meta) when is_map(meta) do
    Enum.reduce(meta, %{}, fn
      {k, v}, acc when is_atom(k) ->
        if k in @allowed_meta_keys, do: Map.put(acc, k, v), else: acc

      {k, v}, acc when is_binary(k) ->
        atom = try_to_atom(k)
        if atom && atom in @allowed_meta_keys, do: Map.put(acc, atom, v), else: acc
    end)
  end

  @doc """
  Attaches the default Chimeway Logger telemetry handler.

  Logs :stop events at :debug level and :exception events at :warning level.
  Safe to call multiple times — subsequent calls are no-ops (catches :already_exists).

  Call this in your Application.start/2:

      def start(_type, _args) do
        Chimeway.Telemetry.attach_default_handlers()
        children = [...]
        Supervisor.start_link(children, strategy: :one_for_one)
      end
  """
  @spec attach_default_handlers() :: :ok
  def attach_default_handlers do
    base_spans = [
      [:chimeway, :events, :create],
      [:chimeway, :deliveries, :plan],
      [:chimeway, :policy, :evaluate],
      [:chimeway, :dispatch, :sync],
      [:chimeway, :attempts, :record]
    ]

    oban_spans =
      if Code.ensure_loaded?(Oban) do
        [[:chimeway, :dispatch, :enqueue], [:chimeway, :dispatch, :perform]]
      else
        []
      end

    all_spans = base_spans ++ oban_spans

    stop_events = Enum.map(all_spans, &(&1 ++ [:stop]))
    exception_events = Enum.map(all_spans, &(&1 ++ [:exception]))

    attach_handler(
      :chimeway_default_logger_stop,
      stop_events,
      fn event, _measurements, metadata, _config ->
        Logger.debug("Chimeway #{inspect(event)}", metadata)
      end
    )

    attach_handler(
      :chimeway_default_logger_exception,
      exception_events,
      fn event, _measurements, metadata, _config ->
        Logger.warning("Chimeway #{inspect(event)} exception", metadata)
      end
    )

    :ok
  end

  defp attach_handler(name, events, handler_fn) do
    case :telemetry.attach_many(name, events, handler_fn, nil) do
      :ok -> :ok
      {:error, :already_exists} -> :ok
    end
  end

  defp try_to_atom(str) do
    String.to_existing_atom(str)
  rescue
    ArgumentError -> nil
  end
end
```

```elixir
defmodule Chimeway.TelemetryIntegrationTest do
  @moduledoc """
  Confirms all mandatory Chimeway telemetry spans fire during a full trigger cycle
  and that no PII keys appear in any stop event's metadata.
  """

  use Chimeway.DataCase, async: false

  alias Chimeway.Telemetry

  @mandatory_stop_events [
    [:chimeway, :events, :create, :stop],
    [:chimeway, :deliveries, :plan, :stop],
    [:chimeway, :policy, :evaluate, :stop],
    [:chimeway, :dispatch, :sync, :stop],
    [:chimeway, :attempts, :record, :stop]
  ]

  @pii_keys [:email, :phone, :body, :payload, :content, :template, :url]

  setup do
    test_pid = self()
    handler_id = :"chimeway_test_handler_#{System.unique_integer()}"

    :telemetry.attach_many(
      handler_id,
      @mandatory_stop_events,
      fn event, _measurements, metadata, _config ->
        send(test_pid, {:telemetry_event, event, metadata})
      end,
      nil
    )

    on_exit(fn -> :telemetry.detach(handler_id) end)

    :ok
  end

  defp run_trigger do
    # Use a notifier that satisfies the Chimeway.Notifier behaviour.
    # Check test/support/ for an existing fixture module; if none, create one.
    # The notifier must implement notification_key/0, version/0, recipients/1, build/2.
    notifier = Chimeway.Test.SupportNotifier
    params = %{user_id: "user:telemetry-#{System.unique_integer()}"}
    opts = [idempotency_key: "telem-key-#{System.unique_integer()}"]
    Chimeway.Trigger.trigger(notifier, params, opts)
  end

  describe "mandatory span emission" do
    test "all 5 mandatory :stop spans fire during a trigger cycle" do
      {:ok, _result} = run_trigger()

      results =
        Enum.map(@mandatory_stop_events, fn expected ->
          receive do
            {:telemetry_event, ^expected, _meta} -> {expected, :ok}
          after
            500 -> {expected, :missing}
          end
        end)

      missing = for {event, :missing} <- results, do: event

      assert missing == [],
             "Expected telemetry events did not fire: #{inspect(missing)}"
    end

    test "no stop event metadata contains PII keys" do
      {:ok, _result} = run_trigger()

      Enum.each(@mandatory_stop_events, fn expected ->
        receive do
          {:telemetry_event, ^expected, metadata} ->
            pii_found = Enum.filter(@pii_keys, &Map.has_key?(metadata, &1))

            assert pii_found == [],
                   "PII keys #{inspect(pii_found)} found in #{inspect(expected)} metadata: #{inspect(metadata)}"
        after
          500 -> flunk("Event #{inspect(expected)} not received within 500ms")
        end
      end)
    end

    test "events:create stop event metadata contains notification_key" do
      {:ok, _result} = run_trigger()

      receive do
        {:telemetry_event, [:chimeway, :events, :create, :stop], metadata} ->
          assert Map.has_key?(metadata, :notification_key),
                 "events:create :stop missing :notification_key. Got: #{inspect(Map.keys(metadata))}"
      after
        500 -> flunk("[:chimeway, :events, :create, :stop] not received")
      end
    end
  end

  describe "safe_meta/1 unit tests" do
    test "drops disallowed atom keys" do
      raw = %{
        notification_key: "order_shipped",
        event_id: "uuid-abc",
        email: "user@example.com",
        phone: "+15551234567",
        payload: %{ssn: "123-45-6789"},
        body: "Dear Customer"
      }

      result = Telemetry.safe_meta(raw)

      assert result == %{notification_key: "order_shipped", event_id: "uuid-abc"}
      refute Map.has_key?(result, :email)
      refute Map.has_key?(result, :phone)
      refute Map.has_key?(result, :payload)
      refute Map.has_key?(result, :body)
    end

    test "drops disallowed string keys" do
      raw = %{
        "notification_key" => "order_shipped",
        "email" => "secret@example.com",
        "delivery_id" => "delivery-uuid-123"
      }

      result = Telemetry.safe_meta(raw)

      assert result == %{notification_key: "order_shipped", delivery_id: "delivery-uuid-123"}
      refute Map.has_key?(result, :email)
      refute Map.has_key?(result, "email")
    end

    test "returns empty map when all keys are disallowed" do
      raw = %{email: "a@b.com", phone: "555", payload: %{}}
      assert Telemetry.safe_meta(raw) == %{}
    end

    test "preserves all allowed keys" do
      allowed = %{
        notification_key: "k",
        event_id: "eid",
        recipient_id: "rid",
        channel: :email,
        delivery_id: "did",
        attempt_id: "aid",
        outcome: :succeeded,
        suppression_reason: "disabled",
```

              Deliveries.record_attempt(dispatched, %{
                outcome: :failed,
                error_class: "temporary",
                provider_response: %{seq: n}

            else

          ordered: false,
          max_concurrency: 5,
          timeout: 15_000

      sorted = Enum.sort(attempt_numbers)
      expected = Enum.to_list(1..length(attempts))

      assert sorted == expected,

    end
  end

  - Final attempt (job.attempt == job.max_attempts) writes :cancelled retries_exhausted

  alias Chimeway.Reliability.TerminalConvergenceTest.{BouncedAdapter, PermanentAdapter}

  setup do
    Application.put_env(:chimeway, :adapter, Chimeway.Adapters.Test)

  describe "exhaust_delivery/1" do
    setup :insert_notification

    test "transitions :failed -> :cancelled with retries_exhausted reason", %{notification: notification} do
      {:ok, delivery} = Deliveries.plan_delivery(notification.id, :in_app)
      {:ok, dispatched} = Deliveries.transition_status(delivery, :dispatched)

      {:ok, %{delivery: failed}} =
        Deliveries.record_attempt(dispatched, %{
          outcome: :failed,
          error_class: "temporary",
          provider_response: %{}
        })

      assert failed.status == :failed

      assert {:ok, exhausted} = Deliveries.exhaust_delivery(failed)
      assert exhausted.status == :cancelled
      assert exhausted.suppression_reason == "retries_exhausted"
      assert exhausted.metadata["policy_checkpoint"] == "perform"
      assert exhausted.status in Deliveries.terminal_states()
    end

    test "rejects exhaust_delivery from :pending status", %{notification: notification} do
      {:ok, delivery} = Deliveries.plan_delivery(notification.id, :in_app)
      assert {:error, {:invalid_exhaust_from, :pending}} = Deliveries.exhaust_delivery(delivery)
    end

    test "rejects exhaust_delivery from :succeeded status", %{notification: notification} do
      {:ok, delivery} = Deliveries.plan_delivery(notification.id, :in_app)
      {:ok, dispatched} = Deliveries.transition_status(delivery, :dispatched)
      {:ok, succeeded} = Deliveries.transition_status(dispatched, :succeeded)

      assert {:error, {:invalid_exhaust_from, :succeeded}} = Deliveries.exhaust_delivery(succeeded)
    end

    test "general transition_status path rejects :failed -> :cancelled (D-10 guard)", %{
      notification: notification
    } do
      {:ok, delivery} = Deliveries.plan_delivery(notification.id, :in_app)
      {:ok, dispatched} = Deliveries.transition_status(delivery, :dispatched)

      {:ok, %{delivery: failed}} =
        Deliveries.record_attempt(dispatched, %{
          outcome: :failed,
          error_class: "temporary",
          provider_response: %{}
        })

      assert {:error, {:invalid_transition, from: :failed, to: :cancelled}} =
               Deliveries.transition_status(failed, :cancelled)
    end
  end
